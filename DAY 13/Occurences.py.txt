 # Count Occurrences
You have:
data = np.array([1, 2, 2, 3, 1, 4, 2, 3])
Task:
● Count how many times each unique number appears.
● Return numbers with their counts.


import numpy as np
# Given data array
data = np.array([1, 2, 2, 3, 1, 4, 2, 3])
# Count occurrences of each unique number
unique_numbers, counts = np.unique(data, return_counts=True)
# Combine numbers with their counts
occurrences = dict(zip(unique_numbers.tolist(), counts.tolist()))
print("Unique numbers:", unique_numbers)
print("Counts:        ", counts)
print("Numbers with counts:", occurrences)
