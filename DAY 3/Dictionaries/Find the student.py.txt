   # Write a program to find the student with the highest marks from a dictionary


student_marks = {
    "Rahul": 85,
    "Ananya": 92,
    "Vikram": 78,
    "Priya": 95
}
top_student = max(student_marks, key=student_marks.get)
print(f"The student with the highest marks is {top_student} with {student_marks[top_student]} marks.")
