    # Write a program to assign grades based on marks (for example: A, B, C, Fail).



marks = int(input("Enter the marks obtained : "))
if marks < 0 or marks > 100:
    print("Invalid marks! Please enter a value between 0 and 100.")
if marks >= 90:
       print ("A")
elif marks >= 80:
       print("B")
elif marks >= 70:
       print("C")
elif marks >= 60:
       print("D")
else:
       print("Fail")
