 # : Develop a Python program for a small shop to process customer purchases. Store product names and prices in a 
dictionary, items added to the cart in a list, product categories in a set,and product details using tuples. Create 
functions to display products, add items to the cart, and calculate the total bill. Use a recursive function to compute 
the total price of all items in the cart.Include exception handling to manage ValueError (invalid quantity input), 
ZeroDivisionError(calculation errors), TypeError (wrong data types in the cart), and NameError (when a product
name entered by the user does not exist).



# Small Shop Billing Management System

# Dictionary: Product names and prices
products = {
    "Rice": 60,
    "Sugar": 45,
    "Oil": 150,
    "Milk": 30,
    "Soap": 40
}

# List: Store cart items
cart = []

# Set: Store product categories
categories = {"Food", "Grocery", "Personal Care"}

# Tuple: Store product details
product_details = (
    ("Rice", "Food"),
    ("Sugar", "Grocery"),
    ("Oil", "Food"),
    ("Milk", "Food"),
    ("Soap", "Personal Care")
)


# Function to display products
def display_products():
    print("\n------ AVAILABLE PRODUCTS ------")

    for name, price in products.items():
        print(name, ":", price)

    print("\nProduct Categories:", categories)


# Function to add items to cart
def add_to_cart():
    try:
        name = input("Enter product name: ").title()

        if name not in products:
            raise NameError("Product does not exist!")

        quantity = int(input("Enter quantity: "))

        if quantity <= 0:
            raise ValueError("Quantity must be greater than zero.")

        price = products[name]

        cart.append((name, price, quantity))

        print(name, "added to cart successfully!")

    except ValueError as e:
        print("Error:", e)

    except NameError as e:
        print("Error:", e)


# Recursive function to calculate total bill
def calculate_total(items):
    try:
        if len(items) == 0:
            return 0

        name, price, quantity = items[0]

        if not isinstance(name, str):
            raise TypeError("Product name must be a string.")

        if not isinstance(price, (int, float)):
            raise TypeError("Price must be numeric.")

        if not isinstance(quantity, int):
            raise TypeError("Quantity must be an integer.")

        return (price * quantity) + calculate_total(items[1:])

    except TypeError:
        raise


# Function to display total bill
def total_bill():
    try:
        if len(cart) == 0:
            print("Your cart is empty.")
            return

        print("\n------ CUSTOMER BILL ------")

        for name, price, quantity in cart:
            print(name, "x", quantity, "=", price * quantity)

        total = calculate_total(cart)

        print("---------------------------")
        print("Total Bill:", total)

    except TypeError as e:
        print("Error:", e)

    except ZeroDivisionError:
        print("Error: Calculation cannot be divided by zero.")


# Main menu
while True:

    print("\n===== SMALL SHOP BILLING SYSTEM =====")
    print("1. Display Products")
    print("2. Add Items to Cart")
    print("3. Calculate Total Bill")
    print("4. Exit")

    try:
        choice = int(input("Enter your choice: "))

        if choice == 1:
            display_products()

        elif choice == 2:
            add_to_cart()

        elif choice == 3:
            total_bill()

        elif choice == 4:
            print("Thank you for shopping!")
            break

        else:
            print("Invalid choice! Select 1 to 4.")

    except ValueError:
        print("Error: Enter a valid menu number.")
