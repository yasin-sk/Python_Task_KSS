 # Filter Values Using Condition
A dataset:
arr = np.array([10, 25, 30, 15, 40])
Task:
● Convert to Pandas Series
● Filter values greater than 20


import numpy as np
import pandas as pd

arr = np.array([10, 25, 30, 15, 40])

# 1. Convert NumPy array to Pandas Series
S = pd.Series(arr)

# 2. Filter values greater than 20
filtered_S = S[S > 20]

print("Filtered Series (values > 20):")
print(filtered_S)
