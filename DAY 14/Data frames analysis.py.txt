 #  Student Marks DataFrame Analysis
A DataFrame:
data = pd.DataFrame({
"Name": ["A", "B", "C"],
"Math": [80, 70, 60],
"Science": [90, 60, 70]
})
Task:
● Add a new column Total = Math + Science
● Find the student with the highest total marks


import pandas as pd

data = pd.DataFrame({
    "Name": ["A", "B", "C"],
    "Math":[80, 70, 60],
    "Science": [90, 60, 70]
})

# 1. Add a new column Total = Math + Science
data["Total"] = data["Math"] + data["Science"]

# 2. Find the student with the highest total marks
top_student = data.loc[data["Total"].idxmax(), "Name"]

print("Updated DataFrame:")
print(data)
print("\nStudent with the highest total marks:", top_student)
