 # Data Alignment Issue in Series Addition
Two Series:
S1 = pd.Series([10, 20, 30], index=["a", "b", "c"])
S2 = pd.Series([5, 15, 25], index=["b", "c", "d"])
Task:
● Add both Series
● Explain why some values become NaN
● Replace NaN with 0 and compute final result


import pandas as pd

S1 = pd.Series([10, 20, 30], index=["a", "b", "c"])
S2 = pd.Series([5, 15, 25], index=["b", "c", "d"])

# 1. Direct addition (produces NaN for non-overlapping indexes)
direct_sum = S1 + S2

# 2. Replace missing values with 0 before adding
final_result = S1.add(S2, fill_value=0)

print("Direct Addition (with NaN):")
print(direct_sum)
print("\nFinal Result (NaN replaced with 0):")
print(final_result)
