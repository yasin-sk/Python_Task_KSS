 # Student Marks Analysis (NumPy → Pandas)
Marks data:
arr = np.array([
[80, 90],
[70, 60],
[85, 95]
])
Task:
● Convert into DataFrame with columns "Math", "Science"
● Add a new column Total
● Find student with highest total


import numpy as np
import pandas as pd

arr = np.array([[85, 95]])

# 1. Convert into DataFrame with columns "Math" and "Science"
df = pd.DataFrame(arr, columns=["Math", "Science"])

# 2. Add a new column Total
df["Total"] = df["Math"] + df["Science"]

# 3. Find student (row index) with highest total
top_student_index = df["Total"].idxmax()

print("Student Marks DataFrame:")
print(df)
print(f"\nStudent with highest total: Index {top_student_index} (Total = {df.loc[top_student_index, 'Total']})")
