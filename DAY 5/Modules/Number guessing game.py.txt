  # Create a Number Guessing Game where:
● The program generates a random number between 1 and 50 using random.
● The user has 5 attempts to guess the number.
● After each guess, calculate the absolute difference using math.fabs() and
display how far the guess is from the correct number.


import math
import random
target_number = random.randint(1, 50)
attempts_allowed = 5
print("Welcome to the Number Guessing Game!")
print("I have chosen a number between 1 and 50.")
print(f"You have {attempts_allowed} attempts to guess it correctly.\n")
for i in range(1, attempts_allowed + 1):
    guess = int(input(f"Attempt {i}/{attempts_allowed} - Enter your guess: "))
    if guess == target_number:
        print(f"🎉 Congratulations! You guessed the correct number ({target_number})!")
        break
    difference = math.fabs(target_number - guess)
    print(f"Wrong guess! You are {difference:.0f} numbers away from the correct answer.\n")
else:
    print("❌ Game Over! You have run out of attempts.")
    print(f"The correct number was: {target_number}")
