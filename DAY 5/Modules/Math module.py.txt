  # Write a Python program using the math module to calculate and display the square root, floor value, and ceiling value 
of a number entered by the user.


import math
number = float(input("Enter a number: "))
square_root = math.sqrt(number)
floor_value = math.floor(number)
ceil_value = math.ceil(number)
print(f"Square root: {square_root}")
print(f"Floor value: {floor_value}")
print(f"Ceiling value: {ceil_value}")
