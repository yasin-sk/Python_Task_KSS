  # Given a string "programming" create a dictionary that counts how many times each charecter appears.


a = "programming"
char_counts = {}
for char in a:
# If the char is already in increase its count 1
    if char in char_counts:
        char_counts[char] += 1
# If it is a new char add it with count of 1
    else:
        char_counts[char] = 1
print(char_counts)
