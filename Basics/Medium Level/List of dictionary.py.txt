employees = [
    {"name": "John", "age": 25, "salary": 30000},
    {"name": "Alice", "age": 30, "salary": 50000},
    {"name": "Bob", "age": 28, "salary": 40000},
    {"name": "David", "age": 35, "salary": 60000}
]
># The max() function uses lambda to look at the 'salary' key inside each dictionary
highest_salary_emp = max(employees, key=lambda emp: emp["salary"])
># Uses a list comprehension to check each employee's salary
high_earners = [emp for emp in employees if emp["salary"] > 35000]
># Sum up all salaries and divide by the total number of employees
total_salary = sum(emp["salary"] for emp in employees)
average_salary = total_salary / len(employees)
># 4. Create a list containing only employee names
employee_names = [emp["name"] for emp in employees]
print(" Employee with highest salary:", highest_salary_emp["name"], f"({highest_salary_emp['salary']})")
print(" Employees with salary > 35000:", [emp["name"] for emp in high_earners])
print(" Average salary:", average_salary)
print(" List of employee names:", employee_names)
