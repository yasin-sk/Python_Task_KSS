 # Shape Area Calculator (Polymorphism)
A graphics application needs to calculate the area of different shapes. Create classes Circle, Rectangle, and Triangle, 
each having an area() method. Demonstrate polymorphism by calling the same method for different objects.


import math


# Shape 1: Circle
class Circle:
    def __init__(self, radius):
        self.radius = radius

    def area(self):
        return math.pi * (self.radius ** 2)


# Shape 2: Rectangle
class Rectangle:
    def __init__(self, length, width):
        self.length = length
        self.width = width

    def area(self):
        return self.length * self.width


# Shape 3: Triangle
class Triangle:
    def __init__(self, base, height):
        self.base = base
        self.height = height

    def area(self):
        return 0.5 * self.base * self.height


# Creating objects of different shape classes
shapes = [
    Circle(5),
    Rectangle(10, 4),
    Triangle(8, 6)
]

# Demonstrating polymorphism by calling the same area() method on each object
print("=== Shape Area Calculator ===\n")
for shape in shapes:
    shape_name = type(shape).__name__
    print(f"{shape_name} Area : {shape.area():.2f}")
