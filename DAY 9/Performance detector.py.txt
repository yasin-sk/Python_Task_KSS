 # Performance Tracker (Decorators)
A software team wants to track how long functions take to execute. Create a decorator that measures and prints the 
execution time of a function.


import time
from functools import wraps


# Decorator function to measure execution time
def measure_time(func):
    @wraps(func)
    def wrapper(*args, **kwargs):
        start_time = time.perf_counter()
        result = func(*args, **kwargs)
        end_time = time.perf_counter()

        elapsed_time = end_time - start_time
        print(f"Function '{func.__name__}' executed in {elapsed_time:.6f} seconds.")
        return result

    return wrapper


# Sample function 1: Processing a large calculation
@measure_time
def calculate_squares(n):
    total = sum(i * i for i in range(n))
    return total


# Sample function 2: Simulating a network or database delay
@measure_time
def fetch_data():
    time.sleep(0.5)
    return "Data fetched successfully"


print("=== Performance Tracker ===\n")

# Calling the decorated functions
sum_result = calculate_squares(1_000_000)
print(f"Calculation Result : {sum_result}\n")

fetch_status = fetch_data()
print(f"Status             : {fetch_status}")
