  # Bank Account System (Class, Object, Constructor)
A bank wants to manage customer accounts. Create a BankAccount class with a constructor to initialize account number and
balance. Implement methods to deposit,withdraw, and display balance.


# Base Class
class Employee:
    def __init__(self, name, salary):
        self.name = name
        self.salary = salary


# Derived Class inheriting from Employee
class Manager(Employee):
    def display_details(self):
        print(f"Employee Name : {self.name}")
        print(f"Salary        : ₹{self.salary:,.2f}")
        print("-" * 30)


# Creating Manager objects
mgr1 = Manager("Yasin", 85000)
mgr2 = Manager("Asif", 105000)

# Displaying details using the derived class method
print("=== Manager Details ===\n")
mgr1.display_details()
mgr2.display_details()
