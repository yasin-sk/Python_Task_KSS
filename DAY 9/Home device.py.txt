 # Smart Home Devices (Multiple Inheritance)
A smart home device may have both WiFi connectivity and Voice control features.Create classes WiFiDevice and 
VoiceAssistant, and a class SmartSpeaker that inherits from both using multiple inheritance.


# Parent Class 1
class WiFiDevice:
    def __init__(self, network_name):
        self.network_name = network_name
        self.is_connected = False

    def connect_wifi(self):
        self.is_connected = True
        print(f"Connected to WiFi network : '{self.network_name}'")


# Parent Class 2
class VoiceAssistant:
    def __init__(self, wake_word):
        self.wake_word = wake_word

    def voice_command(self, command):
        print(f"[{self.wake_word}] Executing command: '{command}'")


# Child Class inheriting from both WiFiDevice and VoiceAssistant
class SmartSpeaker(WiFiDevice, VoiceAssistant):
    def __init__(self, device_name, network_name, wake_word):
        self.device_name = device_name
        WiFiDevice.__init__(self, network_name)
        VoiceAssistant.__init__(self, wake_word)

    def display_status(self):
        status = "Online" if self.is_connected else "Offline"
        print("=== Smart Speaker Details ===")
        print(f"Device Name : {self.device_name}")
        print(f"WiFi Network: {self.network_name} ({status})")
        print(f"Wake Word   : {self.wake_word}")
        print("-" * 30)


# Creating a SmartSpeaker object
speaker = SmartSpeaker("Echo Studio", "Home_Fiber_5G", "Alexa")

# Using methods inherited from both parent classes
speaker.connect_wifi()
speaker.voice_command("Play relaxing instrumental music")
print()

# Displaying full device status
speaker.display_status()
