 # Random Number Analyzer
Scenario:
A system generates random numbers for testing.
Task:
● Use random to generate 10 numbers
● Store in a list
● Use loop + condition to count even/odd numbers
● Use set to remove duplicates


import random

# 1 & 2. Generate 10 random numbers (between 1 and 20) and store in a list
numbers = [random.randint(1, 20) for _ in range(10)]

# 3. Use a loop + condition to count even and odd numbers
even_count = 0
odd_count = 0

for num in numbers:
    if num % 2 == 0:
        even_count += 1
    else:
        odd_count += 1

# 4. Use a set to remove duplicates
unique_numbers = set(numbers)

print("Generated Numbers:", numbers)
print("Even Count:", even_count)
print("Odd Count:", odd_count)
print("Unique Numbers (Set):", unique_numbers)
