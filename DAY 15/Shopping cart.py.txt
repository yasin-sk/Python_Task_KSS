 # Shopping Cart System
Scenario: A user adds items to a shopping cart.
Task:
● Store items in a list
● Convert to set to remove duplicates
● Use loop + condition to calculate total cost
● Handle invalid input using try-except


# Store catalog with item prices
catalog = {
    "apple": 2.0,
    "banana": 1.0,
    "milk": 3.5,
    "bread": 2.5,
    "eggs": 4.0
}

# 1. Store items in a list (including duplicates and invalid items)
cart_list = []

try:
    n = int(input("How many items do you want to add? "))
    if n < 0:
        raise ValueError("Item count cannot be negative.")
        
    for i in range(n):
        item = input(f"Enter item {i + 1} (apple, banana, milk, bread, eggs): ").strip().lower()
        cart_list.append(item)

except ValueError as e:
    print(f"Invalid input! Please enter a valid positive number. ({e})")

# 2. Convert list to set to remove duplicates
unique_cart = set(cart_list)

# 3. Use loop + condition to calculate total cost
total_cost = 0.0
for item in unique_cart:
    if item in catalog:
        total_cost += catalog[item]
    else:
        print(f"Warning: '{item}' is not available in the store and was skipped.")

print("\nRaw Cart (List):", cart_list)
print("Unique Items (Set):", unique_cart)
print(f"Total Cost: ${total_cost:.2f}")
