    #  Write a program using comparison operators.




a = int(input("Enter first number (a): "))
b = int(input("Enter second number (b): "))
print(a, "== ", b, " is ", a == b)
print(a, "!= ", b, " is ", a != b)
print(a, "> ", b, " is ", a > b)
print(a, "< ", b, " is ", a < b)
print(a, ">= ", b, " is ", a >= b)
print(a, "<= ", b, " is ", a <= b)
